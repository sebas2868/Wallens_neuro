cmake_minimum_required(VERSION 3.8)
project(wallens_nav)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(explore_lite_msgs QUIET)   # opcional

add_executable(supervisor_mision src/supervisor_mision.cpp)
target_include_directories(supervisor_mision PRIVATE include)
ament_target_dependencies(supervisor_mision rclcpp std_msgs sensor_msgs nav_msgs slam_toolbox)
if(explore_lite_msgs_FOUND)
  ament_target_dependencies(supervisor_mision explore_lite_msgs)
  target_compile_definitions(supervisor_mision PRIVATE WALLENS_HAS_EXPLORE_LITE_MSGS)
else()
  message(WARNING "explore_lite_msgs no encontrado: el supervisor C++ no leerá explore/status")
endif()
install(TARGETS supervisor_mision DESTINATION lib/${PROJECT_NAME})

install(DIRECTORY config launch behavior_trees DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_pytest QUIET)
  if(ament_cmake_pytest_FOUND)
    ament_add_pytest_test(test_supervisor test/test_supervisor.py)
  endif()
  find_package(ament_cmake_gtest QUIET)
  if(ament_cmake_gtest_FOUND)
    ament_add_gtest(test_supervisor_cpp test/test_supervisor_cpp.cpp)
    target_include_directories(test_supervisor_cpp PRIVATE include)
    target_compile_definitions(test_supervisor_cpp PRIVATE
      CASOS_JSON="${CMAKE_CURRENT_SOURCE_DIR}/test/casos_supervisor.json")
  endif()
endif()
ament_package()
